### 任务 5：完成 MCP 集成项目

**项目要求** 🧠：

创建一个集成了 MCP 工具和基本记忆功能的 Agent：

```python
# mcp_memory_agent.py
"""
MCP + 记忆系统集成项目

功能：
1. 使用 MCP 工具（计算、系统信息）
2. 记住对话历史
3. 可以引用之前的对话内容
"""

import os
from dotenv import load_dotenv
from hello_agents import SimpleAgent, HelloAgentsLLM
from hello_agents.tools import MCPTool

load_dotenv()


class ConversationMemory:
    """简单的对话记忆管理"""
    
    def __init__(self, max_turns=10):
        self.max_turns = max_turns
        self.history = []
    
    def add(self, role: str, content: str):
        """添加一条记忆"""
        self.history.append({"role": role, "content": content})
        # 保持历史在限制内
        if len(self.history) > self.max_turns * 2:
            self.history = self.history[-self.max_turns * 2:]
    
    def get_context(self) -> str:
        """获取记忆上下文"""
        if not self.history:
            return "（暂无历史对话）"
        
        lines = []
        for msg in self.history[-6:]:  # 最近3轮
            role = "用户" if msg["role"] == "user" else "助手"
            lines.append(f"{role}: {msg['content']}")
        
        return "\n".join(lines)


def create_memory_agent():
    """创建带记忆的 MCP Agent"""
    
    llm = HelloAgentsLLM()
    
    # 创建带记忆提示词的 Agent
    agent = SimpleAgent(
        name="智能助手",
        llm=llm,
        system_prompt="""你是一个智能助手，拥有以下能力：
        
1. 数学计算（加减乘除）
2. 系统信息查询
3. 记住对话历史

当前对话历史：
{memory_context}

请根据用户需求选择合适的工具，并在需要时引用之前的对话内容。"""
    )
    
    # 添加 MCP 工具
    mcp_tool = MCPTool(name="builtin")
    agent.add_tool(mcp_tool)
    
    return agent


def main():
    print("=" * 50)
    print("智能助手（MCP + 记忆）")
    print("输入 'quit' 退出，'history' 查看历史")
    print("=" * 50)
    
    agent = create_memory_agent()
    memory = ConversationMemory()
    
    while True:
        question = input("\n你的问题: ").strip()
        
        if question.lower() in ['quit', 'exit', 'q']:
            print("再见！")
            break
        
        if question.lower() == 'history':
            print("\n对话历史:")
            print("-" * 40)
            print(memory.get_context())
            continue
        
        if not question:
            continue
        
        # 添加到记忆
        memory.add("user", question)
        
        # 更新 Agent 的上下文
        agent.system_prompt = agent.system_prompt.replace(
            "{memory_context}", 
            memory.get_context()
        )
        
        print("\n" + "-" * 50)
        response = agent.run(question)
        print("-" * 50)
        print(f"\n回答: {response}")
        
        # 添加回答到记忆
        memory.add("assistant", response)


if __name__ == "__main__":
    main()
```
