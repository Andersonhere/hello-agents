### 任务 6：创建自定义 Agent

**你需要做的事情** 🧠：

使用 HelloAgents 框架创建一个简单的 Agent：

```python
# my_custom_agent.py
from hello_agents import SimpleAgent, HelloAgentsLLM, ToolRegistry

# 1. 创建 LLM 客户端
llm = HelloAgentsLLM()

# 2. 创建工具注册表
registry = ToolRegistry()

# 3. 定义工具函数
def calculator(expression: str) -> str:
    """计算数学表达式"""
    try:
        result = eval(expression)
        return f"计算结果: {expression} = {result}"
    except Exception as e:
        return f"计算错误: {e}"

def get_time(city: str) -> str:
    """获取城市当前时间"""
    from datetime import datetime
    import pytz
    try:
        tz = pytz.timezone(city)
        return f"{city} 当前时间: {datetime.now(tz).strftime('%Y-%m-%d %H:%M:%S')}"
    except:
        return f"无法获取 {city} 的时间"

# 4. 注册工具
registry.register_tool(
    name="calculator",
    func=calculator,
    description="计算数学表达式，如 '1+2*3'"
)

registry.register_tool(
    name="get_time",
    func=get_time,
    description="获取指定城市的当前时间"
)

# 5. 创建 Agent
agent = SimpleAgent(
    name="智能助手",
    llm=llm,
    tool_registry=registry,
    system_prompt="""你是一个智能助手，可以：
1. 计算数学表达式
2. 查询城市时间

请根据用户的问题选择合适的工具。"""
)

# 6. 运行 Agent
if __name__ == "__main__":
    # 测试用例
    test_cases = [
        "帮我算一下 123 * 456",
        "纽约现在几点了？",
        "帮我算 (100 + 200) / 3，然后告诉我东京几点了"
    ]
    
    for question in test_cases:
        print(f"\n问题: {question}")
        print("-" * 40)
        response = agent.run(question)
        print(f"回答: {response}")
```
