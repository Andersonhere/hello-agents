### 任务 3：运行框架示例代码

**LangGraph 基础示例** 🧠：

```python
# langgraph_basic.py
from langgraph.graph import StateGraph, END
from typing import TypedDict, Annotated
import operator
from langchain_openai import ChatOpenAI

# 1. 定义状态
class AgentState(TypedDict):
    messages: Annotated[list, operator.add]
    next_action: str

# 2. 定义节点函数
def agent_node(state: AgentState) -> AgentState:
    """Agent 节点：调用 LLM"""
    llm = ChatOpenAI(model="gpt-3.5-turbo")
    response = llm.invoke(state["messages"])
    return {"messages": [response], "next_action": "end"}

# 3. 构建图
workflow = StateGraph(AgentState)
workflow.add_node("agent", agent_node)
workflow.set_entry_point("agent")
workflow.add_edge("agent", END)

# 4. 编译并运行
app = workflow.compile()
result = app.invoke({"messages": [{"role": "user", "content": "你好"}]})
print(result)
```

### 任务 6：实现 Agent 项目

**完整项目代码示例** 🧠：

```python
# search_agent/main.py
import os
from dotenv import load_dotenv
from hello_agents import SimpleAgent, HelloAgentsLLM, ToolRegistry
from tavily import TavilyClient

load_dotenv()

# 1. 定义搜索工具
def search_tool(query: str) -> str:
    """
    搜索互联网获取信息。
    
    Args:
        query: 搜索关键词或问题
        
    Returns:
        搜索结果摘要
    """
    print(f"🔍 正在搜索: {query}")
    
    tavily = TavilyClient(api_key=os.getenv("TAVILY_API_KEY"))
    result = tavily.search(query, search_depth="basic", max_results=3)
    
    # 整理搜索结果
    summaries = []
    for item in result.get("results", []):
        summaries.append(f"- {item['title']}: {item['content'][:200]}...")
    
    return "\n".join(summaries) if summaries else "未找到相关信息"


# 2. 创建工具注册表
registry = ToolRegistry()
registry.register_tool(
    name="search",
    func=search_tool,
    description="搜索互联网获取实时信息。当你需要回答关于时事、新闻或最新信息时使用此工具。"
)


# 3. 系统提示词
SYSTEM_PROMPT = """你是一个智能搜索助手。

你的能力：
1. 回答各类问题
2. 当问题涉及最新信息或你不确定的事实时，使用搜索工具
3. 整合搜索结果，给出准确、有用的回答

工作流程：
1. 分析用户问题
2. 判断是否需要搜索（涉及实时信息、新闻、你不确定的事实）
3. 如需搜索，使用 search 工具
4. 整合信息，给出完整回答

注意：
- 不要编造信息
- 搜索结果要整合，不要直接复制
- 如果搜索结果不相关，如实告知用户
"""


# 4. 创建 Agent
def create_search_agent():
    llm = HelloAgentsLLM()
    
    agent = SimpleAgent(
        name="智能搜索助手",
        llm=llm,
        tool_registry=registry,
        system_prompt=SYSTEM_PROMPT
    )
    
    return agent


# 5. 主程序
def main():
    print("=" * 50)
    print("智能搜索助手")
    print("输入 'quit' 退出")
    print("=" * 50)
    
    agent = create_search_agent()
    
    while True:
        question = input("\n你的问题: ").strip()
        
        if question.lower() in ['quit', 'exit', 'q']:
            print("再见！")
            break
        
        if not question:
            continue
        
        print("\n" + "-" * 50)
        response = agent.run(question)
        print("-" * 50)
        print(f"\n回答: {response}")


if __name__ == "__main__":
    main()
```
