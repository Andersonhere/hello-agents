```python
# app/agents/travel_agent.py 增强版
import logging
from typing import Optional

logger = logging.getLogger(__name__)

class TravelAgent:
    """智能旅行助手 Agent"""
    
    def __init__(self):
        try:
            self.llm = HelloAgentsLLM()
            self.agent = SimpleAgent(
                name="智能旅行助手",
                llm=self.llm,
                system_prompt=self._get_system_prompt()
            )
            self._setup_tools()
            logger.info("TravelAgent 初始化成功")
        except Exception as e:
            logger.error(f"TravelAgent 初始化失败: {e}")
            raise
    
    def plan(self, destination: str, days: int = 3) -> str:
        """生成旅行计划"""
        if not destination:
            return "请提供有效的目的地"
        
        if days < 1 or days > 30:
            return "天数应在 1-30 天之间"
        
        try:
            logger.info(f"生成旅行计划: {destination}, {days}天")
            prompt = self._build_plan_prompt(destination, days)
            result = self.agent.run(prompt)
            logger.info("旅行计划生成成功")
            return result
        except Exception as e:
            logger.error(f"生成计划失败: {e}")
            return f"抱歉，生成计划时出错: {str(e)}"
    
    def _build_plan_prompt(self, destination: str, days: int) -> str:
        """构建提示词"""
        return f"""
请为以下旅行需求制定详细计划：

目的地：{destination}
天数：{days}天

请提供：
1. 行程安排（每天的主要活动，包含时间安排）
2. 推荐景点（简要介绍和门票信息）
3. 交通建议（如何前往和市内交通）
4. 美食推荐（当地特色美食）
5. 住宿建议（推荐区域和类型）
6. 必备物品清单
7. 旅行注意事项
"""
```

```markdown
- Memory（记忆）

- Planning（规划）

规划 → 分解 → 执行
```
