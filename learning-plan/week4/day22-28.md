#### 4.3 核心Agent

```python
# app/agents/travel_agent.py
from hello_agents import SimpleAgent, HelloAgentsLLM
from hello_agents.tools import MCPTool

class TravelAgent:
    """智能旅行助手 Agent"""
    
    def __init__(self):
        # 创建 LLM
        self.llm = HelloAgentsLLM()
        
        # 创建 Agent
        self.agent = SimpleAgent(
            name="智能旅行助手",
            llm=self.llm,
            system_prompt=self._get_system_prompt()
        )
        
        # 添加 MCP 工具
        self._setup_tools()
    
    def _get_system_prompt(self):
        return """你是一个专业的智能旅行助手。

你的能力：
1. 根据目的地推荐景点和行程
2. 提供旅行建议和注意事项
3. 回答旅行相关问题

请根据用户需求提供专业、实用的旅行建议。"""
    
    def _setup_tools(self):
        """设置工具"""
        # 使用内置 MCP 工具（演示用）
        mcp_tool = MCPTool(name="builtin")
        self.agent.add_tool(mcp_tool)
        
        # 可以添加更多工具
        # 例如：天气工具、景点搜索工具等
    
    def plan(self, destination: str, days: int = 3) -> str:
        """生成旅行计划"""
        prompt = f"""
请为以下旅行需求制定详细计划：

目的地：{destination}
天数：{days}天

请提供：
1. 行程安排（每天的主要活动）
2. 必备物品清单
3. 旅行注意事项
"""
        return self.agent.run(prompt)
    
    def ask(self, question: str) -> str:
        """回答旅行问题"""
        return self.agent.run(question)
```

#### 4.4 API 接口

```python
# app/main.py
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import Optional, List
from datetime import date

from app.agents.travel_agent import TravelAgent

app = FastAPI(title="智能旅行助手 API")

# 初始化 Agent
travel_agent = TravelAgent()


class TravelRequest(BaseModel):
    destination: str
    days: int = 3


class QuestionRequest(BaseModel):
    question: str


@app.get("/")
async def root():
    return {"message": "智能旅行助手 API", "status": "running"}


@app.post("/api/plan")
async def create_plan(request: TravelRequest):
    """生成旅行计划"""
    try:
        plan = travel_agent.plan(request.destination, request.days)
        return {
            "success": True,
            "destination": request.destination,
            "plan": plan
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@app.post("/api/ask")
async def ask_question(request: QuestionRequest):
    """回答旅行问题"""
    try:
        answer = travel_agent.ask(request.question)
        return {
            "success": True,
            "question": request.question,
            "answer": answer
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```
